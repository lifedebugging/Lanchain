import structlog
import sys

# 1. Create a processor that adds file/line location
def add_call_location(_, method_name, event_dict):
    # Skip the internal structlog and processor frames
    try:
        raise Exception()
    except Exception:
        f = sys.exc_info()[2].tb_frame.f_back
        # Walk back until we exit the structlog library
        while f and "structlog" in f.f_code.co_filename:
            f = f.f_back
        
        if f:
            module = f.f_globals.get("__name__")
            lineno = f.f_lineno
            event_dict["location"] = f"{module}:{lineno}"
            
    return event_dict

# 2. Configure structlog to use the processor
structlog.configure(
    processors=[
        add_call_location,
        structlog.processors.add_log_level,
        structlog.processors.JSONRenderer(), # Produces JSON output for easy searching
    ]
)

logger = structlog.get_logger()

# 3. Log using the kwargs you asked for
logger.error(
    "Database connection dropped",
    error_type="TimeoutError",
    message="Connection pool maxed out"
)

# Output includes:
# {"event": "Database connection dropped", "error_type": "TimeoutError", "message": "Connection pool maxed out", "location": "__main__:28", "level": "error"}
